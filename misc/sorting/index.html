    void bubble_sort(int a[], int n)
{
    for (int end = n - 1; end > 0; --end) {
        int changed = 0;
        for (int j = 0; j < end; ++j) {
            if (a[j] > a[j + 1]) {
                int t = a[j]; a[j] = a[j + 1]; a[j + 1] = t;
                changed = 1;
            }
        }
        if (!changed) break;
    }
}

    void selection_sort(int a[], int n)
{
    for (int i = 0; i < n - 1; ++i) {
        int min_index = i;
        for (int j = i + 1; j < n; ++j)
            if (a[j] < a[min_index]) min_index = j;
        int t = a[i]; a[i] = a[min_index]; a[min_index] = t;
    }
}

    void insertion_sort(int a[], int n)
{
    for (int i = 1; i < n; ++i) {
        int value = a[i], j = i;
        while (j > 0 && a[j - 1] > value) {
            a[j] = a[j - 1];
            --j;
        }
        a[j] = value;
    }
}

    /* 快速排序：对闭区间 a[left..right] 排序，
   调用入口为  quick_sort(a, 0, n - 1) */
void quick_sort(int a[], int left, int right)
{
    if (left >= right) return;
    /* 选取 pivot */
    int pivot = a[left + (right - left) / 2];
    /* Hoare 划分 */
    int i = left, j = right;
    while (i <= j) {
        while (a[i] < pivot) ++i;
        while (a[j] > pivot) --j;
        if (i <= j) {
            int t = a[i]; a[i] = a[j]; a[j] = t;
            ++i;
            --j;
        }
    }
    /* 递归排序两部分 */
    if (left < j) quick_sort(a, left, j);
    if (i < right) quick_sort(a, i, right);
}

    /*  归并排序：对 a[left, right)  左闭右开区间排序，
    temp 为与 a 相同长度的辅助数组。
    调用入口为 merge_sort(a, temp, 0, n) */
void merge_sort(int a[], int temp[], int left, int right)
{
    if (right - left <= 1) return;
    int mid = left + (right - left) / 2;
     /* 递归排序左右两半 */
    merge_sort(a, temp, left, mid);
    merge_sort(a, temp, mid, right);
    /* 归并 */
    int i = left, j = mid, k = left;
    while (i < mid && j < right)
        temp[k++] = a[i] <= a[j] ? a[i++] : a[j++];
    while (i < mid) temp[k++] = a[i++];
    while (j < right) temp[k++] = a[j++];
    for (k = left; k < right; ++k) a[k] = temp[k];
}

    /* 向下调整：让 a[root] 在子树中下沉到合适位置，维持大顶堆性质 */
static void sift_down(int a[], int root, int n)
{
    while (root < n / 2) {
        int child = 2 * root + 1;
        if (child + 1 < n && a[child] < a[child + 1])
            ++child;
        if (a[root] >= a[child]) break;
        int t = a[root];
        a[root] = a[child];
        a[child] = t;
        root = child;
    }
}

void heap_sort(int a[], int n)
{
    /* make heap：建大顶堆 */
    for (int root = n / 2; root > 0; --root)
        sift_down(a, root - 1, n);
    /* sort：不断取出堆顶最大值放到数组末尾 */
    for (int end = n - 1; end > 0; --end) {
        int t = a[0];
        a[0] = a[end];
        a[end] = t;
        sift_down(a, 0, end);
    }
}
